# Initialize Masar Environmental Agent
import masar_core as mc
from biophilia_toolkit import organic_growth

class ZenAgent(mc.BaseAgent):
    def __init__(self, environment_state):
        # Set baseline calm parameters
        self.noise_threshold = 0.15
        self.growth_rate = organic_growth.RHYTHMIC_BREATHING
        self.state = environment_state

    def process_stimuli(self, input_data):
        """
        Filters incoming data to reduce cognitive load.
        """
        filtered_data = self._apply_soft_filter(input_data)
        
        if filtered_data.intensity > self.noise_threshold:
            return self._mitigate(filtered_data)
            
        return self._nurture(filtered_data)

    def _nurture(self, data):
        # Emits a soft, diffused response
        response = mc.generate_response(
            type='ambient',
            glow_intensity=0.2
        )
        return response
